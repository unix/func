import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { Value } from 'func'",
  '',
  'class Options {',
  '  @Value()',
  '  tags: string[] = []',
  '}',
].join('\n')

export const fixCode = [
  "import { ArrayValue } from 'func'",
  '',
  'class Options {',
  '  @ArrayValue()',
  '  tags: string[] = []',
  '}',
].join('\n')

## Reproduction

`@Value()` accepts only scalar `String` and `Number` types:

<CodeBlock code={reproductionCode} filename="src/options.ts" lang="ts" />

## How to fix it

Use `@Flag()` for a boolean option, `@ArrayValue()` for repeated `String` or `Number` values, or change the property and explicit `type` to a supported scalar type:
