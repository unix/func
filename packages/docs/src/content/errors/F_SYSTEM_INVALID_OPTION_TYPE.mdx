import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { Value } from 'func'",
  '',
  'class Options {',
  '  @Value()',
  '  tags: string[] = []',
  '}',
].join('\n')

export const fixCode = [
  "import { ArrayString } from 'func'",
  '',
  'class Options {',
  '  @ArrayString()',
  '  tags: string[] = []',
  '}',
].join('\n')

## Reproduction

`@Value()` accepts only properties whose emitted runtime type is `String` or `Number`:

<CodeBlock code={reproductionCode} filename="src/options.ts" lang="ts" />

## How to fix it

Use `@Flag()` for a boolean option, `@ArrayString()` for `string[]`, or `@ArrayNumber()` for `number[]`. Otherwise, change the property declaration to a supported scalar type:

<CodeBlock code={fixCode} filename="src/options.ts" lang="ts" />
