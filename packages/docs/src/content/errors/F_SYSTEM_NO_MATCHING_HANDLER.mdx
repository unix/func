  "import { CommandMajor, Container, Handler } from 'func'",
  '',
  '@CommandMajor()',
  'class RootCommand {',
  "  @Handler({ flag: 'help' })",
  '  help() {}',
  '}',
  '',
  'new Container([RootCommand], { argv: [] }).run()',
].join('\n')

export const fixCode = [
  'class RootCommand {',
  "  @Handler({ flag: 'help' })",
  '  help() {}',
  '',
  '  @Handler()',
  '  run() {}',
  '}',
  '',
  'new Container([RootCommand], { argv: [] }).run()',
].join('\n')
