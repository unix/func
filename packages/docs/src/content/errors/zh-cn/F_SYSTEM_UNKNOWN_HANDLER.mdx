export const reproductionCode = [
  "import { Module, run } from 'func'",
  '',
  'class PlainClass {}',
  '',
  '@Module({ commands: [PlainClass] })',
  'class AppModule {}',
  '',
  'run(AppModule)',
].join('\n')

## 复现

普通类不能直接放入 `Module.commands`：

<CodeBlock code={reproductionCode} filename="src/app.module.ts" lang="ts" />

## 这个错误是什么

注册列表中包含 func 无法归类的类：它不是具名命令、主命令或缺省命令处理器。`details.handlers` 数组会包含所有校验失败的类名。

## 如何修复

为每个已注册命令类添加 `@Command()`、`@CommandMajor()` 或 `@CommandMissing()`。如果这个类是可复用依赖，请用 `@Injectable()` 标记它，并将其移入 `Module.providers`。
