import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { Handler } from 'func'",
  '',
  'class ProjectCommand {',
  "  @Handler({ flag: 'list', path: ['project', 'list'] })",
  '  list() {}',
  '}',
].join('\n')

export const fixCode = [
  'class ProjectCommand {',
  "  @Handler({ path: ['project', 'list'] })",
  '  list() {}',
  '}',
].join('\n')

## 复现

一个处理器只能使用一种选择方式：

<CodeBlock code={reproductionCode} filename="src/project.command.ts" lang="ts" />

## 如何修复

只保留 `flag` 或 `path`：

<CodeBlock code={fixCode} filename="src/project.command.ts" lang="ts" />
