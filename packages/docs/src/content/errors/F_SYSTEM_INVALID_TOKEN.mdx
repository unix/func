import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { Command } from 'func'",
  '',
  "@Command({ name: '-build' })",
  'class BuildCommand {}',
].join('\n')

export const fixCode = ["@Command({ name: 'build' })", 'class BuildCommand {}'].join(
  '\n',
)

## Reproduction

Decorator tokens describe the logical name, so they must not include CLI hyphens or whitespace:

<CodeBlock code={reproductionCode} filename="src/build.command.ts" lang="ts" />

## How to fix it

Pass the token without leading hyphens and use one non-whitespace segment:

<CodeBlock code={fixCode} filename="src/build.command.ts" lang="ts" />
