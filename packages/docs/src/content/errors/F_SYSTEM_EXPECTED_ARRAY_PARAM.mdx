description: A list parameter such as aliases, Enum, DependsOn, or Exclusive is not an array.
lastModified: '2026-08-08'
---

import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { Enum, Value } from 'func'",
  '',
  'class BuildCommand {',
  "  @Enum('production' as any)",
  '  @Value()',
  "  target: string = 'development'",
  '}',
].join('\n')

export const fixCode = [
  'class BuildCommand {',
  "  @Enum(['production'])",
  '  @Value()',
  "  target: string = 'development'",
  '}',
].join('\n')

## Reproduction

Decorator properties that accept a list, including `aliases`, require an array even when the list has one item:

<CodeBlock code={reproductionCode} filename="src/build.command.ts" lang="ts" />

## How to fix it

Wrap the supplied item in an array:

<CodeBlock code={fixCode} filename="src/build.command.ts" lang="ts" />
