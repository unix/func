export const moduleFilterCode = [
  "import { APP_FILTER, Catch, Module } from 'func'",
  "import type { Exception, ExceptionFilter } from 'func'",
  "import { PublishCommand } from './commands/publish.command'",
  '',
  '@Catch()',
  'class AppFilter implements ExceptionFilter {',
  '  catch(exception: Exception) {',
  '    console.error(exception.message)',
  '  }',
  '}',
  '',
  '@Module({',
  '  commands: [PublishCommand],',
  '  providers: [{ provide: APP_FILTER, useClass: AppFilter }],',
  '})',
  'export class AppModule {}',
].join('\n')

export const jsonErrorCode = [
  "import { Catch } from 'func'",

  '    return {',
  '      error: {',

  '        message: exception.message,',
  '      },',
  '    }',
  '  }',

func 使用 Exception Filter 处理跨切面错误，使用 `onInit` 和 `onDispose` 管理资源生命周期。错误分流不再由 module 或 command 的 `onError` 方法承担。

## Exception 与默认行为

业务代码可以抛出原生 `Error`，也可以抛出公开的 `Exception`。`Exception` 提供稳定的 `code`、`scope`、`details`、`message` 和 `cause`，但不定义 JSON wire format；需要 JSON 时应由应用显式挑选字段。

可修正的命令行输入错误和 handler 抛出的普通错误默认设置退出码 `1`，并向 stderr 写入一行普通文本。默认错误路由不添加颜色、边框或表格。系统定义错误、provider 初始化错误等框架故障会继续抛出，除非应用 Filter 明确捕获。

错误 scope 只有 `F_SYSTEM` 与 `F_RUNTIME`。原来的 `F_RUNTIME_PRINT` 已合并到 `F_RUNTIME`。

Filter 正常返回表示错误已处理；抛出新错误会继续进入外层 Filter。Filter 可以直接写 stdout/stderr，作为自定义 UI 的逃生窗口，但默认职责是匹配、分流和产生结构化结果。

## Module 与 AppModule 作用域

使用 DI multi-provider 注册 module 或应用级 Filter：

- `MODULE_FILTER` 只影响直接 owner module。
- `APP_FILTER` 无论在哪个已导入 module 中声明，都会影响整个应用，包括无法选中 command 的解析错误。
- command Filter → owner module Filter → app Filter → func 默认路由。

Filter class 由 DI 创建，因此可以注入当前 module 可见的 providers、`Context` 和标准流。

func 不提供固定错误 JSON schema，也不会自动序列化 `Exception`。Filter 只需返回普通结构化数据；如果当前作用域导入了 `JsonOutputModule`，它会像处理 handler return 一样处理这个结果：

## 生命周期

`onInit` 与 `onDispose` 仍然是资源生命周期，不承担错误分流：

<CodeBlock code={lifecycleCode} filename="src/app.module.ts" lang="ts" />

`onDispose()` 不接收本次调用的结果，返回值也会被忽略。已激活的 command、module 与 provider 会按逆序释放。hook 抛出异常或 Promise reject 时，Runtime 会报告生命周期故障并继续执行剩余 hook，已经确定的 result、exit code 与输出保持不变。生命周期故障不进入 Exception Filter。
