import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { Value } from 'func'",
  '',
  'class Options {',
  '  @Value()',
  '  when: Date = new Date()',
  '}',
].join('\n')

export const fixCode = [
  'class Options {',
  '  @Value()',
  "  when: string = ''",
  '}',
].join('\n')

## 复现

`@Value()` 只能使用 TypeScript 生成的 `String` 和 `Number` 运行时类型：

<CodeBlock code={reproductionCode} filename="src/options.ts" lang="ts" />

## 如何修复

将属性声明为 `string` 或 `number`，让 TypeScript 生成受支持的装饰器元数据：

<CodeBlock code={fixCode} filename="src/options.ts" lang="ts" />

同时确保已启用 `emitDecoratorMetadata`。布尔选项使用 `@Flag()`；重复值分别使用 `string[]` 与 `@ArrayString()`，或 `number[]` 与 `@ArrayNumber()`。
