import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { CommandMajor, Handler, Module, createApp } from 'func'",
  '',
  '@CommandMajor()',
  'class FirstMajor {',
  '  @Handler()',
  '  run() {}',
  '}',
  '',
  '@CommandMajor()',
  'class SecondMajor {',
  '  @Handler()',
  '  run() {}',
  '}',
  '',
  '@Module({ commands: [FirstMajor, SecondMajor] })',
  'class AppModule {}',
  '',
  'createApp(AppModule)',
].join('\n')

export const fixCode = [
  "import { Command, CommandMajor, Handler, Module, createApp } from 'func'",
  '',
  '@CommandMajor()',
  'class FirstMajor {',
  '  @Handler()',
  '  run() {}',
  '}',
  '',
  "@Command({ path: 'second' })",
  'class SecondCommand {',
  '  @Handler()',
  '  run() {}',
  '}',
  '',
  '@Module({ commands: [FirstMajor, SecondCommand] })',
  'class AppModule {}',
  '',
  'createApp(AppModule)',
].join('\n')

## 复现

不以命令名称开头的调用只能由一个命令接管：

<CodeBlock code={reproductionCode} filename="src/app.module.ts" lang="ts" />

## 如何修复

只保留一个主命令，并为另一个命令注册名称：

<CodeBlock code={fixCode} filename="src/app.module.ts" lang="ts" />
