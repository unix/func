import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { Command } from 'func'",
  '',
  "@Command('')",
  'class HelpCommand {}',
].join('\n')

export const fixCode = ["@Command('help')", 'class HelpCommand {}'].join('\n')

## Reproduction

A required command path is empty:

<CodeBlock code={reproductionCode} filename="src/help.command.ts" lang="ts" />

## What this error means

A decorator cannot produce complete command metadata from the supplied parameters. `details.key` identifies the missing value; in the example above it is `path`.

## How to fix it

Supply the missing parameter:

<CodeBlock code={fixCode} filename="src/help.command.ts" lang="ts" />

Command and option names must be non-empty strings. A handler alias without a flag uses the more specific `F_SYSTEM_HANDLER_ALIAS_REQUIRES_FLAG` code.
