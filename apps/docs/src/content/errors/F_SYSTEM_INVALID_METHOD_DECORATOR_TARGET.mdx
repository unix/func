import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { Handler } from 'func'",
  '',
  'class BuildCommand {',
  '  @Handler()',
  '  static run() {}',
  '}',
].join('\n')

export const fixCode = [
  'class BuildCommand {',
  '  @Handler()',
  '  run() {}',
  '}',
].join('\n')

## Reproduction

Method decorators need an instance so func can inject dependencies and invoke the method:

<CodeBlock code={reproductionCode} filename="src/build.command.ts" lang="ts" />

## How to fix it

Use an instance method:

<CodeBlock code={fixCode} filename="src/build.command.ts" lang="ts" />

This error also occurs when `@Deprecated()` decorates a default or path handler. `@Deprecated()` supports `@Handler({ flag })` action options; use the `deprecated` parameter on `@Command` to deprecate a default or path-based command entry point.
