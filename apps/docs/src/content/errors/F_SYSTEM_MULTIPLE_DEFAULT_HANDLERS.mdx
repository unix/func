import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { CommandMajor, Handler, Module, createApp } from 'func'",
  '',
  '@CommandMajor()',
  'class BuildCommand {',
  '  @Handler()',
  '  build() {}',
  '',
  '  @Handler()',
  '  rebuild() {}',
  '}',
  '',
  '@Module({ commands: [BuildCommand] })',
  'class AppModule {}',
  '',
  'createApp(AppModule)',
].join('\n')

export const fixCode = [
  '@CommandMajor()',
  'class BuildCommand {',
  '  @Handler()',
  '  build() {}',
  '',
  "  @Handler({ flag: 'rebuild' })",
  '  rebuild() {}',
  '}',
  '',
  '@Module({ commands: [BuildCommand] })',
  'class AppModule {}',
  '',
  'createApp(AppModule)',
].join('\n')

## Reproduction

func cannot choose between multiple fallback methods:

<CodeBlock code={reproductionCode} filename="src/build.command.ts" lang="ts" />

## How to fix it

Keep one default handler and give every additional handler a flag or path:

<CodeBlock code={fixCode} filename="src/build.command.ts" lang="ts" />
