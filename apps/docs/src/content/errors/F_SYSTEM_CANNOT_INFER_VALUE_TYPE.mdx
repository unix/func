import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { Value } from 'func'",
  '',
  'class Options {',
  '  @Value()',
  '  when: Date = new Date()',
  '}',
].join('\n')

export const fixCode = [
  'class Options {',
  '  @Value()',
  "  when: string = ''",
  '}',
].join('\n')

## Reproduction

`@Value()` can use only the emitted `String` and `Number` runtime types:

<CodeBlock code={reproductionCode} filename="src/options.ts" lang="ts" />

## How to fix it

Declare the property as `string` or `number` so TypeScript emits supported decorator metadata:

<CodeBlock code={fixCode} filename="src/options.ts" lang="ts" />

Ensure `emitDecoratorMetadata` is enabled. For booleans, use `@Flag()`; for repeated values, declare `string[]` with `@ArrayString()` or `number[]` with `@ArrayNumber()`.
