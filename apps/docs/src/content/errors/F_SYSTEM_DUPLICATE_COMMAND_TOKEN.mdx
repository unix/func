---

import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { Command, Handler, Module, createApp } from 'func'",
  '',
  "@Command('build')",
  'class BuildCommand {',
  '  @Handler()',
  '  run() {}',
  '}',
  '',
  "@Command('build')",
  'class RebuildCommand {',
  '  @Handler()',
  '  run() {}',
  '}',
  '',
  '@Module({ commands: [BuildCommand, RebuildCommand] })',
  'class AppModule {}',
  '',
  'createApp(AppModule)',
].join('\n')

export const fixCode = [
  "@Command('build')",
  'class BuildCommand {',
  '  @Handler()',
  '  run() {}',
  '}',
  '',
  "@Command('rebuild')",
  'class RebuildCommand {',
  '  @Handler()',
  '  run() {}',
  '}',
  '',
  '@Module({ commands: [BuildCommand, RebuildCommand] })',
  'class AppModule {}',
  '',
  'createApp(AppModule)',
].join('\n')

## Reproduction

Canonical path tokens and aliases share one namespace under the same parent path:

<CodeBlock code={reproductionCode} filename="src/commands.ts" lang="ts" />

## How to fix it

Keep every token unique under its parent path, or move one Command to a different path:

<CodeBlock code={fixCode} filename="src/commands.ts" lang="ts" />
