---
code: F_SYSTEM_INVALID_OPTION_ALIAS
title: Invalid option alias
description: A field option, option command, or handler flag contains an alias that is not exactly one character.
lastModified: '2026-08-08'
---

import CodeBlock from '@/components/content/code-block.astro'

export const reproductionCode = [
  "import { Flag } from 'func'",
  '',
  'class Options {',
  "  @Flag({ aliases: ['v', 'verbose'] })",
  '  verbose = false',
  '}',
].join('\n')

export const fixCode = [
  'class Options {',
  "  @Flag({ aliases: ['v', 'V'] })",
  '  verbose = false',
  '}',
].join('\n')

## Reproduction

Aliases represent short options and must contain exactly one character:

<CodeBlock code={reproductionCode} filename="src/options.ts" lang="ts" />

## How to fix it

Use one character for every entry or remove the invalid alias:

<CodeBlock code={fixCode} filename="src/options.ts" lang="ts" />
